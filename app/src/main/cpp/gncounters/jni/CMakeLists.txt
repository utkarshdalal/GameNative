cmake_minimum_required(VERSION 3.13.5)
project(gncounters CXX)
set(HWCPIPE_WERROR OFF CACHE BOOL "" FORCE)
set(HWCPIPE_ENABLE_LTO OFF CACHE BOOL "" FORCE)
set(HWCPIPE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(HWCPIPE_FRONTEND_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(HWCPIPE_LOGGER_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/gn_logger.h" CACHE STRING "" FORCE)
add_subdirectory(${LIBGPUCOUNTERS_DIR} ${CMAKE_BINARY_DIR}/libgpucounters)
add_library(gncounters SHARED gncounters.cpp)
set_target_properties(gncounters PROPERTIES CXX_STANDARD 17 CXX_VISIBILITY_PRESET hidden)
target_compile_options(gncounters PRIVATE -fno-exceptions -fno-rtti -ffunction-sections -fdata-sections)
target_link_libraries(gncounters PRIVATE hwcpipe device log)
target_link_options(gncounters PRIVATE -Wl,-z,max-page-size=16384 -Wl,--gc-sections -Wl,--build-id=none -Wl,--exclude-libs,ALL)
